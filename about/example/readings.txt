Stlc.Step
  (\lambda{:}A.\, M)\; V \longrightarrow M[V]
    reads only as  Stlc.Step[Stlc.Tm.app[Stlc.Tm.lam[A; M]; V]; Stlc.subst[M; V]]
  M_{1} \longrightarrow M_{1}'
    reads only as  Stlc.Step[M_{1}; M_{1}']
  M_{1}\; M_{2} \longrightarrow M_{1}'\; M_{2}
    reads only as  Stlc.Step[Stlc.Tm.app[M_{1}; M_{2}]; Stlc.Tm.app[M_{1}'; M_{2}]]
  M_{2} \longrightarrow M_{2}'
    reads only as  Stlc.Step[M_{2}; M_{2}']
  V\; M_{2} \longrightarrow V\; M_{2}'
    reads only as  Stlc.Step[Stlc.Tm.app[V; M_{2}]; Stlc.Tm.app[V; M_{2}']]

Stlc.Tm
  \mathit{Term}
    reads only as  Stlc.Tm
  x_{i}
    reads only as  Stlc.Tm.var[i]
  \lambda{:}A.\, M
    reads only as  Stlc.Tm.lam[A; M]
  M_{1}\; M_{2}
    reads only as  Stlc.Tm.app[M_{1}; M_{2}]
  ()
    reads only as  Stlc.Tm.unit
