\LeanProse{Let $M \in \LeanMathConst{TransitiveGround}$ and $K \in V$. Assume \LeanHyp{pairing} $M \models \mathsf{Pair}$, \LeanHyp{unions} $M \models \mathsf{Union}$, \LeanHyp{powers} $M \models \mathsf{Power}$, \LeanHyp{separation} $M \models \mathsf{Sep}$, \LeanHyp{collection} $M \models \mathsf{Coll}$, \LeanHyp{infinite} $M \models \mathsf{Inf}$, \LeanHyp{choice} $M \models \mathsf{AC}$, \LeanHyp{small} $M\ \text{countable}$, \LeanHyp{hK} $K \in M$ and \LeanHyp{standard} $K\ \text{inaccessible in}\ M$. Then there is $N$ such that (1)~$M \subseteq N$, (2)~$N \models \mathsf{ZF}$, (3)~$N \models \mathsf{PP}$ and (4)~$N \not\models \mathsf{AC}$.}
