\LeanProse{Let $M \in \LeanMathConst{TransitiveGround}$ and $K \in V$. Assume $M \models \mathsf{Pair}$, $M \models \mathsf{Union}$, $M \models \mathsf{Power}$, $M \models \mathsf{Sep}$, $M \models \mathsf{Coll}$, $M \models \mathsf{Inf}$, $M \models \mathsf{AC}$, $M\ \text{countable}$, $K \in M$ and $K\ \text{inaccessible in}\ M$. Then there is $N$ such that (1)~$M \subseteq N$, (2)~$N \models \mathsf{ZF}$, (3)~$N \models \mathsf{PP}$ and (4)~$N \not\models \mathsf{AC}$.}
